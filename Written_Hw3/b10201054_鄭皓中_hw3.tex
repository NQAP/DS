\section{Amortized Analysis}
First we named the stack S, with its size $|S|=n$.
\begin{enumerate}
    \item[(a)] The worst case time complexity of MultiPop is $\min(n, k)$. We know that the time complexity of Pop is $O(1)$. MultiPop(k) repeats $\min(n, k)$ times Pop, hence the time complexity of MultiPop(k) is $\min(n, k) \times O(1) = O(\min(n, k))$
    \item[(b)] Suppose that there are $n_{push}, n_{pop}, n_{multipop}$ times of Push, Pop, MultiPop to execute. Note that the legal Pop or MultiPop can only execute when the stack is not empty, we can find that the Push operation needs to be executed same as or more than Pop + MultiPop.
    That is, $n_{pop} + n_{multipop}\leq n_{push}$.\\
    The total time complexity is $T(n) = n_{push} + n_{pop} + n_{multipop} \leq 2\cdot n_{push} = O(n)$\\
    Hence, the amortized time complexity of an operation is $O(n)/n = O(1)$
\end{enumerate}

\section{Algorithm Design}

\begin{algorithm}
    \caption{Find successor}
    \KwIn{x, T.root}
    \KwOut{the successor of x}
    n = T.root
    succ = None
    \While{n is not None}
    {
        \uIf{$x < n.value$}
        {
            succ = n.value\\
            n = n.left
        }
        \uElseIf{$x \geq n.value$}
        {
            n = n.right
        }
    }
    \uIf{succ is not None}
    {
        \Return succ
    }
    \uElse 
    {
        \Return "cannot find successor in the tree"
    }
\end{algorithm}
